function openFreeModal() { /* Détecte mobile ou desktop */ const isMobile = /iPhone|iPad|iPod|Android/i.test(navigator.userAgent); const isIOS = /iPhone|iPad|iPod/i.test(navigator.userAgent); if (isMobile) { /* Redirige directement vers le bon store */ if (isIOS) { window.open('https://apps.apple.com/app/fideliio', '_blank'); } else { window.open('https://play.google.com/store/apps/fideliio', '_blank'); } } else { /* Sur desktop : affiche un petit popup avec les deux stores */ const html = `
🍟

Commencer gratuitement

Téléchargez l'application Fideliio et créez votre compte commerçant en 2 minutes. C'est gratuit !

Télécharger sur
App Store
Disponible sur
Google Play
`; document.body.insertAdjacentHTML('beforeend', html); } }
